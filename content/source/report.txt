โรงเรียนและผู้ประกอบการรายย่อย โครงงานนี้จึงพัฒนาเครื่องตรวจจับและระบุพิกัดตำแหน่งรอยร้าวแบบเรียลไทม์

เพียโซอิเล็กทริก ซึ่งเปลี่ยนคลื่นกลเป็นสัญญาณไฟฟ้าด้วยการเลื่อนประจุในโครงผลึกระดับนาโนเมตร ติดตั้งบนแผ่น
วัสดุทดสอบ 4 ตำแหน่ง ต่อกับไมโครคอนโทรลเลอร์ STM32 ที่บันทึกรูปคลื่นดิบด้วยอัตรา 444,000 ตัวอย่างต่อ
วินาที แล้วหาเวลามาถึงของคลื่นด้วยเทคนิคตรวจจับจุดเริ่มคลื่น (first-break detection) แทนการจับเวลาที่ระดับ
กระตุ้น จากนั้นนําผลตางเวลามาถึงระหวางคูเซนเซอรมาคํานวณพิกัดรวมกับความเร็วคลื่นที่สอบเทียบไว
ผลการทดลองพบว่าคลื่นที่แผ่ในแผ่นวัสดุเป็นคลื่นดัด ความเร็วประมาณ 680 เมตรต่อวินาที ต่ำกว่าค่าคลื่นอัดตาม
ตำราเกือบสิบเท่า จึงต้องสอบเทียบในวัสดุจริงเสมอ เมื่อทดสอบระบุตำแหน่งในแนวหนึ่งมิติที่จุดอ้างอิง 5 จุด
ระหว่างเซนเซอร์คู่ที่ห่างกัน 30 เซนติเมตร จุดละ 3 ครั้ง ระบบให้ความคลาดเคลื่อนเฉลี่ย 0.7 เซนติเมตร (สูงสุด
1.8 เซนติเมตร) คิดเป็นร้อยละ 2 ของระยะระหว่างเซนเซอร์ ปัจจุบันอยู่ระหว่างขยายผลสู่การระบุพิกัดสองมิติ ผล

โครงสร้างทางวิศวกรรมที่อยู่รอบตัวเรา ทั้งอาคาร สะพาน เสาส่งไฟฟ้า และท่อส่งของเหลว ล้วนเสื่อมสภาพตาม
เวลาจากการรับน้ำหนักซ้ำ ๆ การเปลี่ยนแปลงอุณหภูมิ และการกัดกร่อน ความเสียหายเหล่านี้ไม่ได้เกิดขึ้นทันที
ทันใด แต่เริ่มต้นจากรอยร้าวขนาดเล็กระดับไมโครเมตรที่มองไม่เห็นด้วยตาเปล่า แล้วค่อย ๆ ขยายตัวจนถึงจุดที่
โครงสร้างวิบัติ
ปัญหาสำคัญคือวิธีตรวจสอบที่ใช้กันอยู่ทั่วไปในประเทศไทยยังพึ่งพาการตรวจด้วยสายตาเป็นหลัก ซึ่งจะพบความ
เสียหายก็ต่อเมื่อรอยร้าวขยายจนมองเห็นได้แล้ว ซึ่งมักสายเกินกว่าจะป้องกัน ส่วนวิธีตรวจสอบแบบไม่ทำลายที่มี
ความละเอียดสูง เช่น การทดสอบด้วยคลื่นอัลตราโซนิกหรือการถ่ายภาพด้วยรังสี ต้องหยุดการใช้งานโครงสร้าง ต้อง
มีผูเชี่ยวชาญ และตรวจไดทีละจุด จึงไมเหมาะกับการเฝาระวังอยางตอเนื่อง
เทคนิคอะคูสติกอิมิชชัน (Acoustic Emission: AE) มีลักษณะที่แตกต่างจากวิธีอื่นอย่างชัดเจน เพราะเป็นวิธีเชิงรับ
กล่าวคือระบบไม่ได้ส่งพลังงานเข้าไปในวัสดุ แต่ทำหน้าที่เพียง "ฟัง" คลื่นที่วัสดุปล่อยออกมาเองในขณะที่รอยร้าว
กำลังขยายตัว ผลที่ตามมาคือ AE เฝ้าติดตามโครงสร้างได้ตลอดเวลาขณะใช้งานจริง ครอบคลุมพื้นที่กว้างด้วย
เซนเซอร์ไม่กี่ตัว และตรวจจับเฉพาะรอยร้าวที่กำลังลุกลามอยู่ ซึ่งเป็นรอยร้าวที่มีความเสี่ยงจริง ต่างจากวิธีอื่นที่
ตรวจพบรอยบกพร่องทุกชนิดโดยไมแยกวากําลังขยายตัวหรือหยุดนิ่งแลว
อย่างไรก็ตาม ระบบ AE เชิงพาณิชย์มีราคาตั้งแต่หลักแสนถึงหลักล้านบาท เกินกำลังของโรงเรียน ผู้ประกอบการ
รายย่อย และหน่วยงานท้องถิ่นที่ดูแลโครงสร้างสาธารณะจำนวนมาก ช่องว่างนี้เองที่ทำให้เทคโนโลยีซึ่งมีประโยชน์
มากกลับเขาไมถึงผูที่ตองการใชจริง
คณะผู้จัดทำจึงตั้งคำถามว่า หลักการทำงานของระบบ AE ระดับอุตสาหกรรมสามารถย่อลงมาสร้างเองด้วยอุปกรณ์
ที่หาซื้อได้ทั่วไปในราคาหลักพันบาทได้หรือไม่ และหากทำได้ ระบบที่สร้างขึ้นจะระบุตำแหน่งของแหล่งกำเนิดคลื่น
ได้แม่นยำเพียงใด โครงงานนี้จึงพัฒนาต้นแบบเครื่องตรวจจับและระบุตำแหน่งความเสียหายในวัสดุ โดยใช้เซนเซอร์

อะคูสติกอิมิชชัน (Acoustic Emission: AE) หมายถึง คลื่นความดันที่วัสดุปล่อยออกมาเองขณะเกิดการเสียรูปหรือ

ผลต่างเวลาที่คลื่นมาถึง (TDOA) หมายถึง ผลต่างของเวลาที่คลื่นเดินทางจากแหล่งกำเนิดเดียวกันไปถึงเซนเซอร์สอง

การตรวจจับจุดเริ่มคลื่น (First-break Detection) หมายถึง วิธีหาเวลามาถึงของคลื่น โดยหาจุดแรกที่สัญญาณเบี่ยง

2.1 อะคูสติกอิมิชชันและการประยุกตใชในงานวิศวกรรม
อะคูสติกอิมิชชัน (Acoustic Emission: AE) คือปรากฏการณ์ที่วัสดุปล่อยพลังงานออกมาในรูปของคลื่นความดัน
เมื่อได้รับแรงกระทำจนเกิดการเสียรูปหรือรอยร้าวภายใน กลไกพื้นฐานคือ เมื่อรอยร้าวขยายตัว พันธะระหว่าง
อะตอมบริเวณปลายรอยร้าวจะขาดออกอย่างฉับพลัน พลังงานความเครียดที่สะสมไว้จึงถูกปลดปล่อยออกมาส่วน
หนึ่งในรูปของคลื่นยืดหยุ่นที่แผ่ไปทั่วเนื้อวัสดุ แหล่งกำเนิด AE แบ่งได้เป็น 2 ประเภท  แหล่งกำเนิดปฐมภูมิ
(Primary AE Sources) เกิดจากการเปลี่ยนแปลงโครงสร้างภายในวัสดุโดยตรง ได้แก่ การก่อตัวและการแพร่
กระจายของรอยร้าว การเสียรูปถาวร (Plastic Deformation) การเคลื่อนที่ของดิสโลเคชัน และการแตกหักของ
เส้นใยในวัสดุผสม แหล่งกำเนิดทุติยภูมิ (Secondary AE Sources) เกิดจากปรากฏการณ์ที่ตามมาหลังความเสีย
หายเริ่มต้น ได้แก่ การเสียดสีระหว่างผิวรอยร้าวที่เปิด-ปิด การรั่วซึมของของไหลผ่านรอยแตก และการหลุดร่อนของ
ชั้นเคลือบผิว ในงานวิศวกรรมโครงสร้าง เทคนิค AE มีข้อได้เปรียบสำคัญ 3 ประการ ประการแรกคือตรวจจับได้
แบบเรียลไทม์ขณะโครงสร้างยังใช้งานอยู่ โดยไม่ต้องหยุดการทำงาน ประการที่สองคือครอบคลุมพื้นที่ขนาดใหญ่ได้
ด้วยเซนเซอร์จำนวนน้อย เพราะคลื่นเดินทางได้ไกลผ่านเนื้อวัสดุ ประการที่สามคือไวต่อความเสียหายระยะเริ่มต้น
จึงเตือนภัยล่วงหน้าได้ก่อนที่ความเสียหายจะลุกลามถึงขั้นวิกฤต ข้อจำกัดที่สำคัญของ AE คือความอ่อนไหวต่อ
สัญญาณรบกวนจากสิ่งแวดล้อม ความจำเป็นในการสอบเทียบกับวัสดุแต่ละชนิดก่อนใช้งาน และความยากในการ
แยกแยะว่าสัญญาณที่ตรวจพบมาจากรอยราวจริงหรือจากแหลงอื่น
2.2 การเปรียบเทียบวิธีตรวจสอบแบบไมทําลายและเหตุผลที่เลือกอะคูสติกอิมิชชัน
การตรวจสอบแบบไม่ทําลาย (Non-Destructive Testing: NDT) มีหลายวิธี แตละวิธีมีขอดีและขอจํากัดตางกัน
การทดสอบด้วยคลื่นอัลตราโซนิก (Ultrasonic Testing) ยิงคลื่นเสียงความถี่สูงเข้าไปในวัสดุแล้ววัดคลื่นสะท้อน
จากรอยบกพร่อง ให้ความละเอียดสูงและระบุขนาดรอยร้าวได้ดี แต่ต้องเลื่อนหัววัดไปทั่วชิ้นงานทีละจุด จึงใช้เวลา
มากเมื่อพื้นที่กว้าง การถ่ายภาพด้วยรังสี (Radiographic Testing) ให้ภาพภายในวัสดุที่ชัดเจน แต่ต้องใช้แหล่ง
กำเนิดรังสีซึ่งมีข้อกำหนดด้านความปลอดภัยสูง ค่าใช้จ่ายมาก และไม่เหมาะกับการเฝ้าติดตามต่อเนื่อง การตรวจ
ด้วยสารแทรกซึมและกระแสไหลวน (Penetrant / Eddy Current Testing) ตรวจได้เฉพาะรอยบกพร่องที่ผิวหรือ
ใกล้ผิว และโดยทั่วไปต้องหยุดใช้งานโครงสร้างเพื่อเข้าตรวจ ข้อแตกต่างเชิงหลักการที่สำคัญที่สุด คือ AE เป็นวิธีเชิง
รับ (Passive) ระบบไม่ได้ส่งพลังงานเข้าไปในวัสดุ แต่รับฟังคลื่นที่วัสดุปล่อยออกมาเอง ขณะที่วิธีอื่นทั้งหมดเป็นวิธี
เชิงรุก (Active) ที่ต้องส่งพลังงานเข้าไปตรวจ ผลที่ตามมาจากความแตกต่างนี้คือ AE ตรวจจับเฉพาะรอยร้าวที่กำลัง
ขยายตัวอยู่ ซึ่งเป็นรอยร้าวที่มีความเสี่ยงจริง ต่างจากวิธีอื่นที่ตรวจพบรอยบกพร่องทุกชนิดโดยไม่แยกว่ากำลัง
ลุกลามหรือหยุดนิ่งแล้ว ในทางกลับกัน AE ก็ไม่พบรอยร้าวที่หยุดขยายตัวแล้ว และระบุขนาดหรือรูปร่างของรอย
ร้าวไม่ได้โดยตรง ในทางปฏิบัติจึงนิยมใช้ AE เป็นระบบเฝ้าระวังเพื่อคัดกรองตำแหน่งที่น่าสงสัย แล้วใช้อัลตราโซนิก
หรือรังสีตรวจซ้ำเฉพาะตำแหน่งนั้นเพื่อประเมินขนาดความเสียหาย ด้วยเหตุนี้โครงงานนี้จึงเลือกใช้หลักการ AE

2.4 โหมดคลื่น Lamb และการกระจายตามความถี่
คลื่น Lamb แบ่งออกเป็น 2 กลุ่มโหมดตามลักษณะการสั่นของแผ่น โหมดสมมาตร 
(Symmetric mode: S0, S1, …) ผิวทั้งสองด้านของแผ่นเคลื่อนที่สวนทางกัน ทำให้แผ่นยืดและหดในแนวระนาบ
โหมด S0 มีความเร็วสูงและค่อนข้างคงที่ในย่านความถี่ต่ำ โหมดไม่สมมาตร (Antisymmetric mode: A0, A1, …)
ผิวทั้งสองด้านเคลื่อนที่ไปทางเดียวกัน ทำให้แผ่นโค้งงอขึ้นลง จึงเรียกอีกชื่อว่าโหมดดัด (Flexural mode) โหมด
A0 มีความเร็วต่ำกว่า S0 มาก และเปลี่ยนแปลงตามความถี่อย่างชัดเจน คุณสมบัติที่ความเร็วขึ้นอยู่กับความถี่นี้
เรียกว่า การกระจาย (Dispersion) และเป็นลักษณะเฉพาะที่สำคัญที่สุดของคลื่นในแผ่นบาง ความเร็วเฟสของคลื่น
ดัดโดยประมาณคือ C ∝ √ω กล่าวคือ องคประกอบความถี่สูงเดินทางเร็วกวาองคประกอบความถี่ตํ่า
ผลของการกระจายต่อการวัด มี 2 ประการ
ประการแรก คลื่นลูกเดียวกันที่ประกอบด้วยหลายความถี่จะค่อย ๆ แยกตัวออกจากกันขณะเดินทาง ทำให้รูปคลื่นที่
เซนเซอร์ตัวไกลรับได้กว้างและลาดกว่าที่เซนเซอร์ตัวใกล้ การกำหนดจุดอ้างอิงเวลาจึงต้องใช้เกณฑ์เดียวกันในทุก
ช่องสัญญาณ มิฉะนั้นผลตางเวลาที่วัดไดจะคลาดเคลื่อนเชิงระบบ
ประการที่สอง วิธีกระตุ้นที่ต่างกันซึ่งให้ย่านความถี่ต่างกันจะทำให้วัดความเร็วคลื่นได้ต่างกัน ข้อสรุปเชิงปฏิบัติที่
สำคัญคือ การสอบเทียบความเร็วคลื่นต้องทำด้วยแหล่งกำเนิดชนิดเดียวกับที่ใช้วัดจริงเสมอ ในทางปฏิบัติ การเคาะ
ที่ผิวแผ่นในแนวตั้งฉากจะกระตุ้นโหมด A0 เป็นหลัก เพราะเป็นการทำให้แผ่นโค้งงอโดยตรง ดังนั้นความเร็วคลื่นที่
วัดได้จากการทดลองในโครงงานนี้จึงเป็นความเร็วของโหมด A0 ไม่ใช่ความเร็วคลื่นตามยาว 1,640 เมตรต่อวินาทีที่
คำนวณจากคาคงที่ของวัสดุ และเปนเหตุผลที่คาที่วัดไดจริงตํ่ากวาคาเชิงทฤษฎีอยางมีนัยสําคัญ
2.3 คลื่นความดันในวัสดุแข็งและคลื่นนําในแผนบาง
เมื่อเกิดการแตกร้าวในวัสดุแข็ง พลังงานที่ปล่อยออกมาจะแพร่กระจายในรูปของคลื่นความเค้น (Stress Wave) ใน
ตัวกลางกวางไมจํากัด คลื่นแบงเปน 2 ชนิดหลัก ไดแก
คลื่นตามยาว (P-wave หรือ Longitudinal Wave) อนุภาคของวัสดุสั่นในทิศเดียวกับการเดินทางของคลื่น มี
ความเร็วสูงที่สุดจึงมาถึงเซนเซอรกอนเสมอ ความเร็วโดยประมาณคํานวณไดจาก
v = √(E / ρ) โดยที่ E คือมอดุลัสของยัง (หน่วยพาสคัล) และ ρ คือความหนาแน่นของวัสดุ (กิโลกรัมต่อลูกบาศก์
เมตร) คลื่นเฉือน (S-wave หรือ Shear Wave) อนุภาคสั่นในทิศตั้งฉากกับการเดินทาง มีความเร็วต่ำกว่าคลื่นตาม
ยาวประมาณครึ่งหนึ่ง สำหรับอะคริลิก (PMMA) ที่ใช้ในโครงงานนี้ มีมอดุลัสของยังประมาณ 3.2 × 109 พาสคัล
และความหนาแน่นประมาณ 1,190 กิโลกรัมต่อลูกบาศก์เมตร แทนค่าในสมการได้ความเร็วคลื่นตามยาวเชิงทฤษฎี
ประมาณ 1,640 เมตรต่อวินาที
อย่างไรก็ตาม ค่านี้ใช้กับกรณีของโครงงานนี้ไม่ได้โดยตรง เพราะเมื่อความหนาของแผ่นน้อยเมื่อเทียบกับ
ความยาวคลื่น คลื่นจะสะท้อนไปมาระหว่างผิวบนและผิวล่างจนซ้อนทับกันกลายเป็นคลื่นนำ (Guided Wave) ซึ่งมี
พฤติกรรมต่างจากคลื่นในตัวกลางกว้างไม่จำกัดอย่างสิ้นเชิง คลื่นนำในแผ่นบางเรียกว่าคลื่น Lamb ซึ่งอธิบายใน
หัวข้อถัดไป นอกจากนี้ ระหว่างการเดินทาง แอมพลิจูดของคลื่นจะลดลงตามระยะทางจาก 2 กลไก ได้แก่ การกระ
จายพลังงานเชิงเรขาคณิต (Geometric Spreading) ซึ่งพลังงานเท่าเดิมกระจายบนพื้นที่ที่กว้างขึ้น และการดูดกลืน
พลังงานของวัสดุ (Material Attenuation) ซึ่งพลังงานกลบางส่วนเปลี่ยนเป็นความร้อน ทั้งสองกลไกทำให้เซนเซอร์

สัญญาณเดินทางมาถึงเซนเซอร์แต่ละตัว ใช้กันแพร่หลายตั้งแต่ระบบระบุตำแหน่งด้วยดาวเทียม ระบบตรวจจับเรือ
ดำนํ้า ไปจนถึงการหาศูนยกลางแผนดินไหว
หลักการทางเรขาคณิตคือ เมื่อคลื่นเดินทางจากแหล่งกำเนิดที่ตำแหน่ง (x₀, y₀) ไปยังเซนเซอร์ที่ตำแหน่ง (xᵢ, yᵢ)
ระยะทางคือ
dᵢ = √[(x₀ − xᵢ)² + (y₀ − yᵢ)²]
หากทราบความเร็วคลื่น v ผลตางเวลาที่คลื่นมาถึงเซนเซอรคูใดคูหนึ่งจะสัมพันธกับผลตางระยะทางตามสมการ
dᵢ − dⱼ = v × Δtᵢⱼ
จุดสำคัญคือ ผลต่างเวลาไม่ได้บอกตำแหน่ง แต่บอกผลต่างระยะทาง และเซตของจุดทั้งหมดบนระนาบที่มีผลต่าง
ระยะทางถึงจุดคงที่สองจุดเท่ากันคือเส้นไฮเพอร์โบลา ดังนั้นเซนเซอร์หนึ่งคู่จึงบอกได้เพียงว่าแหล่งกำเนิดอยู่ที่ใดที่
หนึ่งบนเสนโคงนั้น ไมสามารถระบุจุดเดียวได
เมื่อเพิ่มเซนเซอร์คู่ที่สองที่วางในแนวต่างออกไป จะได้ไฮเพอร์โบลาอีกเส้นหนึ่ง จุดตัดของไฮเพอร์โบลาทั้งสองเส้น
คือตำแหน่งของแหล่งกำเนิด นี่คือเหตุผลเชิงเรขาคณิตที่ทำให้ระบบระบุตำแหน่ง 2 มิติต้องใช้เซนเซอร์อย่างน้อย 3
ตัว และเปนเหตุผลที่โครงงานนี้ใช 4 ตัว
สำหรับระบบ 4 เซนเซอร โดยกําหนดใหเซนเซอรที่คลื่นมาถึงกอนเปนตัวอางอิง จะสรางสมการได 3 สมการ ดังนี้
d₂ − d₁ = v × Δt₂₁
d₃ − d₁ = v × Δt₃₁
d₄ − d₁ = v × Δt₄₁
2.6 การระบุตำแหน่งด้วยผลต่างเวลามาถึง
เมื่อเกิดเหตุการณ์ที่จุดใดจุดหนึ่งบนแผ่นวัสดุ คลื่นจะแผ่ออกจากจุดนั้นไปทุกทิศทางด้วยอัตราเร็วเท่ากัน เซนเซอร์ที่

หากทราบอัตราเร็วของคลื่นในวัสดุ (v) และวัดผลต่างของเวลาที่คลื่นเดินทางไปถึงเซนเซอร์สองตัวได้ (Δt) ก็จะ
ทราบผลต่างของระยะทางทันที

จุดกำเนิดคลื่นจึงต้องเป็นจุดที่มีผลต่างระยะทางถึงเซนเซอร์ทั้งสองตัวเท่ากับ Δd พอดี ซึ่งตามนิยามทางเรขาคณิต

โฟกัส

2.9 การเจือจางความแม่นยำเชิงเรขาคณิต

Precision หรือ GDOP)
สาเหตุอยู่ที่มุมตัดของเส้นไฮเพอร์โบลา เมื่อจุดกำเนิดอยู่ภายในกรอบเซนเซอร์ เส้นไฮเพอร์โบลาจะตัดกันเป็นมุมชัน

เซนเซอร์ โดยเฉพาะบริเวณมุมทแยงที่ยื่นออกไป เส้นไฮเพอร์โบลาจะเกือบขนานกัน ความคลาดเคลื่อนของเวลาเท่า

ปรากฏการณ์เพียโซอิเล็กทริกคือสมบัติของผลึกบางชนิดที่ผลิตประจุไฟฟ้าขึ้นบนผิวเมื่อถูกแรงกล และในทางกลับ
กันจะเปลี่ยนรูปเมื่อได้รับสนามไฟฟ้า ในการตรวจจับคลื่นเสียงเราใช้สมบัติในทิศทางแรก คือเมื่อคลื่นเดินทางมาถึง
และทำใหผิววัสดุสั่น แผนเพียโซที่ติดอยูจะเปลี่ยนรูปตามและผลิตแรงดันไฟฟาออกมา
โครงงานนี้ใช้แผ่นเพียโซชนิดที่ใช้ทำออดไฟฟ้า ซึ่งราคาถูกและหาซื้อได้ทั่วไป ข้อจำกัดคือมีความถี่กำทอนของตัวเอง
และไม่ได้ออกแบบมาให้ตอบสนองแบนราบตลอดย่านความถี่ ต่างจากเซนเซอร์ AE เชิงพาณิชย์ที่ผ่านการสอบเทียบ
มาแล้ว
คุณภาพของการยึดเซนเซอร์มีผลอย่างมากต่อสัญญาณ เนื่องจากคลื่นต้องถ่ายเทพลังงานจากผิววัสดุเข้าสู่ตัว
เซนเซอร์ผ่านรอยต่อ หากรอยต่อไม่แนบสนิทจะเกิดการสูญเสียพลังงานและความหน่วง การทดลองในโครงงานนี้พบ
ว่าความหนวงที่เกิดจากการยึดมีคาถึง 17 ถึง 20 ไมโครวินาที และเปลี่ยนไปทุกครั้งที่ถอดเซนเซอรออกมาติดใหม
2.12 การแปลงสัญญาณแอนะล็อกเปนดิจิทัลและทฤษฎีบทการสุมตัวอยาง
ไมโครคอนโทรลเลอรอานคาแรงดันดวยวงจรแปลงสัญญาณแอนะล็อกเปนดิจิทัล (ADC) ความละเอียด 12 บิต ซึ่ง
แบงชวงแรงดันอางอิง 3.3 โวลตออกเปน 4,096 ระดับ ทําให 1 ระดับมีคาเทากับ 805.66 ไมโครโวลต
การสุมสัญญาณตอเนื่องใหเปนคาที่จุดเวลาไมตอเนื่องมีขอจํากัดพื้นฐานที่เรียกวา ทฤษฎีบทการสุมตัวอยางของไนค
วิสต ซึ่งระบุวาอัตราการสุมตองมากกวาสองเทาของความถี่สูงสุดที่ตองการวัด มิฉะนั้นความถี่สูงจะปรากฏเปน
ความถี่ตํ่าปลอมในขอมูล เรียกวาปรากฏการณแอเลียส
ระบบนี้อานสัญญาณ 4 ชองแบบวนรอบ ใชเวลารอบละ 26.0 ไมโครวินาที จึงไดอัตราการสุม 38,462 ตัวอยางตอ
วินาทีตอชอง และมีขีดจํากัดไนควิสตที่ 19,231 เฮิรตซ
ตัวเลขนี้เปนขอจํากัดสําคัญที่ตองระบุอยางตรงไปตรงมา เพราะคลื่น AE จากรอยราวจริงในทางวิชาการมีความถี่
ตั้งแตประมาณ 20 กิโลเฮิรตซขึ้นไป ซึ่งอยูเหนือขีดจํากัดของฮารดแวรที่สรางขึ้น สิ่งที่ระบบนี้ตรวจจับไดจริงจึง
เปนการตอบสนองแบบคลื่นดัดโคงของแผนวัสดุตอเหตุการณ ไมใชคลื่น AE ความถี่สูงโดยตรง
2.13 การเข้าถึงหนวยความจําโดยตรงและบัฟเฟอรวงกลม
หากให้หน่วยประมวลผลอ่านค่าจาก ADC ทีละค่าแล้วเก็บลงหน่วยความจำเอง จะเสียเวลาไปกับงานนี้จนไม่เหลือ
เวลาทำอยางอื่น และจังหวะการอานจะไมสมํ่าเสมอเมื่อมีงานอื่นแทรก
การเข้าถึงหน่วยความจำโดยตรง (Direct Memory Access หรือ DMA) แก้ปัญหานี้โดยให้วงจรเฉพาะทางทำหน้าที่
ย้ายข้อมูลจาก ADC ไปยังหน่วยความจำโดยไม่รบกวนหน่วยประมวลผล ทำให้ได้จังหวะการสุ่มที่สม่ำเสมอและ
หน่วยประมวลผลวางพอจะทํางานอื่น
โครงงานนี้ใช้ DMA ร่วมกับ บัฟเฟอร์วงกลม (Circular Buffer) ขนาด 16,384 ตัวอย่าง ซึ่งเมื่อเขียนจนเต็มจะวน
กลับมาเขียนทับตําแหนงแรก ผลคือในหนวยความจําจะมีขอมูลยอนหลังลาสุดอยูเสมอ
ข้อได้เปรียบสำคัญคือ ระบบสามารถเก็บสัญญาณช่วงก่อนเกิดเหตุการณ์ได้ เมื่อตรวจพบว่าสัญญาณเกินเกณฑ์
ระบบจะย้อนกลับไปเอาข้อมูล 512 รอบก่อนหน้าและเก็บต่ออีก 1,536 รอบ รวมเป็น 2,048 รอบต่อเหตุการณ์

2.14 การตรวจจับเหตุการณ์ด้วยเส้นฐานปรับตัว
การกำหนดเกณฑ์ตรวจจับเป็นค่าคงที่ตายตัวมีข้อเสียคือ เมื่อสภาพแวดล้อมเปลี่ยนไปหรือเส้นฐานของสัญญาณ
เลื่อน ระบบจะทริกเกอรผิดพลาดหรือไมทริกเกอรเลย
ระบบนี้จึงคำนวณเส้นฐานแบบปรับตัวตามเวลา ของแต่ละช่องอย่างต่อเนื่อง แล้วเปรียบเทียบค่าที่อ่านได้กับเส้น
ฐานนั้นแทนการเทียบกับศูนย์ นอกจากนี้ยังมีกลไก ประตูความเงียบ (Quiet Gate) ซึ่งกำหนดว่าระบบจะรับ
เหตุการณ์ใหม่ได้ก็ต่อเมื่อสัญญาณสงบลงจนต่ำกว่าเกณฑ์แล้วเท่านั้น เพื่อป้องกันการทริกเกอร์ซ้ำจากคลื่นสะท้อนที่
ยังสั่นคางอยู
2.15 สัญญาณรบกวนและวงจรกราวด
การวัดสัญญาณระดับไมโครโวลตทําใหระบบไวตอสัญญาณรบกวนทางไฟฟามาก แหลงรบกวนที่พบในโครงงานนี้
และมีผลรุนแรงที่สุดคือ วงจรกราวด (Ground Loop) ที่เกิดจากอะแดปเตอรจายไฟแบบสองขาของคอมพิวเตอร
เมื่ออุปกรณสองตัวที่ตอกันมีจุดอางอิงกราวดคนละจุดและมีความตางศักยเล็กนอยระหวางกัน จะเกิดกระแสไหลวน
ในสายกราวด ซึ่งปรากฏเปนสัญญาณรบกวนซอนทับบนสัญญาณที่ตองการวัด
การทดลองแบบสลับเงื่อนไขไปมาสองรอบยืนยันผลอยางชัดเจน เมื่อเสียบอะแดปเตอร ระดับสัญญาณรบกวนเพิ่ม
จาก 7 เปน 22 ถึง 75 ระดับ และเกิดการทริกเกอรปลอม 52 และ 49 ครั้งในเวลา 30 วินาที เมื่อถอดอะแดปเตอร
ออก การทริกเกอรปลอมเปนศูนยทั้งสองรอบ
2.16 การหาจุดเริ่มคลื่น
จุดเริ่มคลื่น (First Break) คือเวลาที่คลื่นเดินทางมาถึงเซนเซอร์จริง ซึ่งเป็นค่าที่วิธีผลต่างเวลามาถึงต้องการ การหา
ค่านี้ให้แม่นยำเป็นหัวใจของความแม่นยำทั้งระบบ เพราะความคลาดเคลื่อน 15 ไมโครวินาทีจะกลายเป็น
ความคลาดเคลื่อนของตําแหนงประมาณ 1 เซนติเมตร
วิธีที่ง่ายที่สุดคือใช้เกณฑ์คงที่ แต่มีข้อเสียร้ายแรงที่เรียกว่า การเลื่อนของเกณฑ์ (Threshold Walk) คือสัญญาณที่
แรงกว่าจะข้ามเกณฑ์เร็วกว่าสัญญาณที่อ่อนกว่า ทั้งที่คลื่นมาถึงพร้อมกัน ทำให้เซนเซอร์ที่อยู่ไกลกว่าดูเหมือนได้รับ
คลื่นชากวาความเปนจริง
โครงงานนี้แก้ปัญหาด้วยการใช้ เกณฑ์ที่เป็นสัดส่วนของยอดคลื่นในช่องนั้นเอง คือร้อยละ 3 ของค่าสูงสุดที่ช่องนั้น
วัดได้ พร้อมกำหนดพื้นของเกณฑ์ไว้ที่ 8 เท่าของระดับสัญญาณรบกวนเพื่อป้องกันการตรวจจับสัญญาณรบกวนเป็น
จุดเริ่มคลื่น วิธีนี้ทําใหทุกชองวัดที่ตําแหนงเดียวกันบนรูปคลื่นของตนเอง จึงไมเกิดการเลื่อนของเกณฑ

2.18 หนาตางวิเคราะหและการรั่วไหลของสเปกตรัม

เพียงชวงหนึ่ง เรียกวา หนาตางวิเคราะห การตัดที่ขอบทําใหเกิดความไมตอเนื่อง ซึ่งปรากฏในสเปกตรัมเปนพลังงาน

นี้ใช

คลื่นจริงมาก ทําใหชวงเงียบครองพื้นที่สวนใหญและเกิดสิ่งประดิษฐความถี่ตํ่าในสเปกตรัม เมื่อรวมกับการคัดกรอง
ขอมูลดวยเกณฑแอมพลิจูดของยอดสเปกตรัม จะเกิดความลําเอียงเชิงความถี่ที่ทําใหผลสรุปผิดพลาดอยางมีนัย
สําคัญ ระบบที่ปรับปรุงแลวจึงใชหนาตางที่ปรับความกวางตามซองของคลื่นจริงในแตละชอง

3.3.1 การจัดวางเซนเซอรและการตอวงจร

คณะผู้จัดทำเห็นว่าข้อจำกัดนี้ต้องระบุไว้อย่างตรงไปตรงมา เนื่องจากมีผลต่อการตีความว่าระบบนี้ทำอะไรได้จริง
อย่างไรก็ตาม ข้อจำกัดดังกล่าวไม่ได้ทำให้หลักการระบุตำแหน่งใช้ไม่ได้ เพราะวิธีผลต่างเวลามาถึงอาศัยเพียงผลต่าง
ของเวลาที่คลื่นเดินทางถึงเซนเซอร์แต่ละตัว ไม่ได้ขึ้นกับว่าคลื่นนั้นมีความถี่เท่าใด ดังที่ผลการทดลองที่ 1 ยืนยันว่า

5.2.6 การจำแนกระดับพลังงาน
การกำหนดเกณฑ์จำแนกจากช่วงไดนามิกของฮาร์ดแวร์ ทำให้เส้นแบ่งระหว่างระดับมีที่มาที่อธิบายได้จากคุณสมบัติ

5.2.7 การเปรียบเทียบกับงานวิจัยที่เกี่ยวของ

/* ae_node.c  -  STM32F411CEU6  4-channel AE front end */
#include "main.h"
#define NCH   4
#define NBUF  4096
#define FS    38462          /* samples per second per ch */
#define PRE   256            /* pre-trigger samples       */
static uint16_t buf[NBUF][NCH];
static float    base[NCH], thr[NCH];
static volatile uint32_t wr = 0;
extern ADC_HandleTypeDef hadc1;
extern TIM_HandleTypeDef htim2;
void HAL_ADC_ConvHalfCpltCallback(ADC_HandleTypeDef *h){ wr = NBUF/2; }
void HAL_ADC_ConvCpltCallback(ADC_HandleTypeDef *h){ wr = NBUF; }
static void learn_baseline(void){
  for (int c = 0; c < NCH; c++){
    uint32_t s = 0;
    for (int i = 0; i < 512; i++) s += buf[i][c];
    base[c] = s / 512.0f;
    thr[c]  = 8.0f * noise_rms(c);      /* 8x noise floor */
  }
}
int main(void){
  HAL_Init(); SystemClock_Config();
  MX_GPIO_Init(); MX_DMA_Init(); MX_ADC1_Init();
  MX_TIM2_Init(); MX_USB_DEVICE_Init();
  HAL_ADC_Start_DMA(&hadc1, (uint32_t*)buf, NBUF*NCH);
  HAL_TIM_Base_Start(&htim2);           /* sample clock @ FS */
  HAL_Delay(200); learn_baseline();
  while (1){
    for (int c = 0; c < NCH; c++){
      if (fabsf(buf[wr-1][c] - base[c]) > thr[c]){
        HAL_Delay(20);                  /* wait for all ch */
        send_event(buf, wr - PRE, 1024);
        HAL_Delay(100);                 /* dead time */
        break;
      }
    }
  }
}

# ae_locate.py  -  first break, TDOA grid search, LOO check
import numpy as np, itertools
FS   = 38462
DT   = 1.0 / FS
V    = 680.0                       # m/s  (calibrated A0)
SENS = np.array([[0,0],[30,0],[30,30],[0,30]], float) / 100.0
def first_break(x, k=0.03, nfloor=8.0):
    x   = x - x[:200].mean()
    thr = max(k * np.abs(x).max(), nfloor * x[:200].std())
    return int(np.argmax(np.abs(x) > thr)) * DT
def grid_search(toa, idx, step, x0=0.0, y0=0.0, span=0.30):
    ref  = idx[int(np.argmin(toa))]
    dt   = toa - toa.min()
    xs   = np.arange(x0, x0 + span + step, step)
    ys   = np.arange(y0, y0 + span + step, step)
    best = (1e9, 0.0, 0.0)
    for x in xs:
        for y in ys:
            d    = np.hypot(SENS[idx,0] - x, SENS[idx,1] - y)
            pred = (d - np.hypot(*(SENS[ref] - [x, y]))) / V
            e    = float(((pred - dt) ** 2).sum())
            if e < best[0]: best = (e, x, y)
    return best[1], best[2]
def leave_one_out(toa):
    sols = []
    for k in range(4):
        keep = [i for i in range(4) if i != k]
        sols.append(grid_search(toa[keep], keep, 0.002))
    sols   = np.array(sols)
    spread = max(np.hypot(*(a - b))
                 for a, b in itertools.combinations(sols, 2))
    return sols.mean(0), spread
def solve(frames):
    toa  = np.array([first_break(ch) for ch in frames])
    x, y = grid_search(toa, [0,1,2,3], 0.005)
    _, spread = leave_one_out(toa)
    amp  = max(ch.ptp() for ch in frames)
    lvl  = ("Critical" if amp > 113.6 else
            "Warning"  if amp > 102.9 else "Minor")
    return dict(x=x*100, y=y*100, level=lvl,
                trusted=bool(spread <= 0.03))   # 3 cm criterion
